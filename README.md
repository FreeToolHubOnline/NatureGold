async function uploadToStorage(file, folder){
  if(!storage) throw new Error('Firebase Storage is not initialized.');
  const cleanName=(file.name||'image').replace(/[^a-zA-Z0-9._-]/g,'_');
  const fileRef=storage.ref(`${folder}/${Date.now()}_${cleanName}`);
  await fileRef.put(file);
  return await fileRef.getDownloadURL();
}

  const q=Math.max(1,Math.min(20,Number(document.getElementById('quantity').value)||1));
  const price=Number(selectedProduct.price);
  const method=document.getElementById('paymentMethod').value;
  if(!Number.isFinite(price)||price<=0){toast('Product price is invalid.');return;}

  const order={
    customerName:document.getElementById('customerName').value.trim(),
    phone:document.getElementById('customerPhone').value.trim(),
    address:document.getElementById('customerAddress').value.trim(),
    city:document.getElementById('customerCity').value.trim(),
    product:selectedProduct.name,
    weight:selectedProduct.weight||'',
    quantity:q,
    unitPrice:price,
    total:price*q,
    paymentMethod:method,
    status:'New',
    createdAt:firebase.firestore.FieldValue.serverTimestamp()
  };

  try{
    await db.collection('Orders').add(order);
    toast('Order received! NatureGold will contact you to confirm.');
    e.target.reset();
    document.getElementById('paymentInfo').classList.remove('show');
    closeModal('checkoutModal');
  }catch(err){
    console.error(err);
    toast('Order save nahi hua. Firestore Rules check karein.');
  }
});

document.getElementById('productForm').addEventListener('submit',async e=>{
  e.preventDefault();
  if(!currentUser){toast('Admin login required.');return;}
  const id=document.getElementById('editProductId').value;
  const existingUrl=id?(products.find(x=>x.id===id)?.imageUrl||''):'';
  const finalImageUrl=uploadedProductUrl||document.getElementById('pImage').value.trim()||existingUrl;

  const data={
    name:document.getElementById('pName').value.trim(),

    price:Number(document.getElementById('pPrice').value),

    resetProductForm();

async function loadAdminProducts(){
  const list=document.getElementById('adminProductList');
  list.textContent='Loading…';
  try{
    const snap=await db.collection('Products').get();
    list.innerHTML='';
    if(snap.empty){list.textContent='Abhi koi product nahi hai.';return;}
    snap.forEach(doc=>{
      const p=doc.data();
      const wrap=document.createElement('div');
      wrap.className='admin-product';
      const img=document.createElement('img');
      img.src=p.imageUrl||FALLBACK_IMAGE;
      img.onerror=()=>img.src=FALLBACK_IMAGE;

      const meta=document.createElement('div');
      meta.className='meta';
      const title=document.createElement('b');
      title.textContent=p.name||'Honey';
      const desc=document.createElement('div');
      desc.textContent=(p.weight||'')+' • '+money(p.price);
      meta.append(title,desc);

      const actions=document.createElement('div');
      actions.className='admin-actions';

      const edit=document.createElement('button');
      edit.className='btn sm';
      edit.textContent='EDIT';
      edit.onclick=()=>{
        document.getElementById('editProductId').value=doc.id;
        document.getElementById('pName').value=p.name||'';
        document.getElementById('pWeight').value=p.weight||'';
        document.getElementById('pPrice').value=p.price||'';
        document.getElementById('pCategory').value=p.category||'Honey';
        document.getElementById('pImage').value=p.imageUrl||'';
        uploadedProductUrl=p.imageUrl||'';
        const pv=document.getElementById('photoPreview');
        if(p.imageUrl){pv.src=p.imageUrl;pv.style.display='block';}
        else{pv.src='';pv.style.display='none';}
        document.getElementById('pPhoto').value='';
        document.getElementById('pDescription').value=p.description||'';
        document.getElementById('saveProductBtn').textContent='SAVE CHANGES';
        document.getElementById('productForm').scrollIntoView({behavior:'smooth'});
      };

      const del=document.createElement('button');
      del.className='btn danger sm';
      del.textContent='DELETE';
      del.onclick=async()=>{
        if(!confirm('Is product ko delete karna hai?')) return;
        try{
          await db.collection('Products').doc(doc.id).delete();
          toast('Product deleted.');
          await loadProducts();
          await loadAdminProducts();
        }catch(err){
          console.error(err);
          toast('Delete nahi hua.');
        }
      };

      actions.append(edit,del);
      wrap.append(img,meta,actions);
      list.append(wrap);
    });
  }catch(e){
    console.error(e);
    list.textContent='Products load nahi hue. Rules check karein.';
  }
}

/* Admin Orders */
async function loadOrders(){
  const list=document.getElementById('ordersList');
  if(!currentUser) return;
  list.textContent='Loading orders…';
  try{
    const snap=await db.collection('Orders').orderBy('createdAt','desc').limit(100).get();
    list.innerHTML='';
    if(snap.empty){list.textContent='Abhi koi order nahi aaya.';return;}
    snap.forEach(doc=>{
      const o=doc.data();
      const wrap=document.createElement('div');
      wrap.className='order';

      const info=document.createElement('div');
      info.textContent=`${o.customerName||''} | ${o.phone||''} | ${o.city||''}\nAddress: ${o.address||''}\n${o.product||''} (${o.weight||''}) x${o.quantity||1} | ${money(o.total)} | ${o.paymentMethod||''}`;
      info.style.whiteSpace='pre-line';

      const row=document.createElement('div');
      row.style.display='flex';
      row.style.gap='8px';
      row.style.marginTop='10px';
      row.style.flexWrap='wrap';

      const status=document.createElement('select');
      ['New','Confirmed','Dispatched','Delivered','Cancelled'].forEach(s=>{
        const op=document.createElement('option');
        op.value=s;
        op.textContent=s;
        if((o.status||'New')===s) op.selected=true;
        status.append(op);
      });

      const save=document.createElement('button');
      save.className='btn sm';
      save.textContent='UPDATE STATUS';
      save.onclick=async()=>{
        try{
          await db.collection('Orders').doc(doc.id).update({status:status.value});
          toast('Order status updated.');
        }catch(e){
          console.error(e);
          toast('Status update nahi hua.');
        }
      };

      row.append(status,save);
      wrap.append(info,row);
      list.append(wrap);
    });
  }catch(e){
    console.error(e);
    list.textContent='Orders load nahi hue. Firestore Rules check karein.';
  }
}
